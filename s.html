Project Overview

Mission Statement

FIDC Sentinel is the first onchain oracle that monitors Brazilian securitization fund (FIDC) health using real government data, computes six credit quality metrics under cryptographic consensus via Chainlink CRE, and emits cross-chain risk alerts when thresholds are breached.

Problem Statement

  • Brazil's R$700B FIDC market prices illiquid credit assets with minimal external verification
  • CVM Resolution 175 opened FIDCs to retail investors — 138,520 individuals now exposed
  • Banco Master/FIDC High Tower scandal: 2,300% reported returns through layered fraud
  • No machine-readable, tamper-resistant health signal exists for these funds

Solution

  • CRE workflow fetches real CVM CSV data daily (zero-auth, public government data)
  • Computes 6 health metrics under BFT consensus by DON nodes
  • Writes immutable health scores to Sepolia with cryptographic proof
  • CCIP broadcasts CRITICAL alerts cross-chain to Arbitrum Sepolia

src/main.ts - Workflow Entry Point

import { Runner, HTTPClient, EVMClient } from "@chainlink/cre-sdk";
import { z } from "zod";
import { fetchCVMData } from "./fetchers/cvm";
import { fetchCDIRate } from "./fetchers/bcb";
import { verifyCNPJ } from "./fetchers/brasilapi";
import { computeHealthScore } from "./metrics/healthScore";
import type { FIDCHealthReport, Config } from "./types";

// Zod schema for config validation
const ConfigSchema = z.object({
  fidcCnpjs: z.array(z.string()),
  oracleAddress: z.string(),
  receiverAddress: z.string(),
  sepoliaRpcUrl: z.string(),
  arbitrumRpcUrl: z.string(),
  ccipRouterAddress: z.string(),
  dataFeedAddress: z.string(),  // BRL/USD
});

export default Runner.newRunner<z.infer<typeof ConfigSchema>>()
  .withConfigSchema(ConfigSchema)
  .withHandler(async (runtime, config, trigger) => {
    // 1. Fetch external data
    const cvmData = await runtime.sendRequest(
      async (http: HTTPClient) => fetchCVMData(http, config.fidcCnpjs)
    );

    const cdiRate = await runtime.sendRequest(
      async (http: HTTPClient) => fetchCDIRate(http)
    );

    const managerStatuses = await runtime.sendRequest(
      async (http: HTTPClient) =>
        Promise.all(cvmData.map(f => verifyCNPJ(http, f.managerCnpj)))
    );

    // 2. Compute health metrics for each FIDC
    const reports: FIDCHealthReport[] = cvmData.map((fidc, i) =>
      computeHealthScore(fidc, cdiRate, managerStatuses[i])
    );

    // 3. Generate and submit reports on-chain
    for (const report of reports) {
      // Write to FIDCHealthOracle.sol
      await runtime.report({
        targetChain: "ethereum-sepolia",
        targetAddress: config.oracleAddress,
        data: {
          cnpj: report.cnpj,
          healthScore: report.healthScore,
          riskLevel: report.riskLevel,
          subordinationRatio: report.subordinationRatio,
          defaultRate: report.defaultRate,
          timestamp: Date.now(),
        },
      });

      // 4. If CRITICAL, send CCIP alert to Arbitrum
      if (report.riskLevel === "CRITICAL") {
        await runtime.sendCCIP({
          destinationChain: "arbitrum-sepolia",
          receiver: config.receiverAddress,
          data: {
            sourceCnpj: report.cnpj,
            healthScore: report.healthScore,
            alertType: "CRITICAL_HEALTH",
          },
        });
      }
    }

    return { processed: reports.length, criticalAlerts: reports.filter(r => r.riskLevel === "CRITICAL").length };
  });

src/fetchers/cvm.ts - CVM Data Fetcher

import type { HTTPClient } from "@chainlink/cre-sdk";
import type { CVMFIDCData } from "../types";

const CVM_BASE_URL = "https://dados.cvm.gov.br/dados/FIDC/DOC/INF_MENSAL/DADOS";

export async function fetchCVMData(
  http: HTTPClient,
  cnpjs: string[]
): Promise<CVMFIDCData[]> {
  // Get latest monthly file (format: inf_mensal_fidc_YYYYMM.csv)
  const now = new Date();
  const yearMonth = `${now.getFullYear()}${String(now.getMonth()).padStart(2, '0')}`;
  const url = `${CVM_BASE_URL}/inf_mensal_fidc_${yearMonth}.csv`;

  const response = await http.get(url, {
    headers: { "Accept": "text/csv" },
  });

  if (!response.ok) {
    // Fallback to previous month
    const prevMonth = `${now.getFullYear()}${String(now.getMonth() - 1).padStart(2, '0')}`;
    const fallbackUrl = `${CVM_BASE_URL}/inf_mensal_fidc_${prevMonth}.csv`;
    const fallbackResp = await http.get(fallbackUrl);
    if (!fallbackResp.ok) throw new Error(`CVM fetch failed: ${fallbackResp.status}`);
    return parseCSV(fallbackResp.body, cnpjs);
  }

  return parseCSV(response.body, cnpjs);
}

function parseCSV(csv: string, filterCnpjs: string[]): CVMFIDCData[] {
  const lines = csv.split('\n');
  const headers = lines[0].split(';');

  // Key column indices (based on CVM schema)
  const cnpjIdx = headers.indexOf('CNPJ_FUNDO');
  const plIdx = headers.indexOf('VL_PATRIM_LIQ');           // AUM
  const subordinadaIdx = headers.indexOf('VL_COTA_SUBORDINADA');
  const inadimpIdx = headers.indexOf('VL_CARTEIRA_INADIMPLENTE');
  const carteiraIdx = headers.indexOf('VL_CARTEIRA_TOTAL');
  const pddIdx = headers.indexOf('VL_PROVISAO');
  const gestorIdx = headers.indexOf('CNPJ_GESTOR');
  const rentabIdx = headers.indexOf('RENTAB_MES');

  return lines.slice(1)
    .map(line => line.split(';'))
    .filter(cols => filterCnpjs.includes(cols[cnpjIdx]))
    .map(cols => ({
      cnpj: cols[cnpjIdx],
      aum: parseFloat(cols[plIdx]) || 0,
      subordinatedValue: parseFloat(cols[subordinadaIdx]) || 0,
      defaultedCredits: parseFloat(cols[inadimpIdx]) || 0,
      totalPortfolio: parseFloat(cols[carteiraIdx]) || 0,
      pddProvisions: parseFloat(cols[pddIdx]) || 0,
      managerCnpj: cols[gestorIdx],
      monthlyReturn: parseFloat(cols[rentabIdx]) || 0,
    }));
}

src/metrics/healthScore.ts - Composite Score Calculator

import type { CVMFIDCData, FIDCHealthReport, RiskLevel } from "../types";

const WEIGHTS = {
  subordination: 0.25,
  defaultRate: 0.25,
  pddCoverage: 0.20,
  concentration: 0.15,
  yieldSpread: 0.10,
  managerStatus: 0.05,
};

export function computeHealthScore(
  fidc: CVMFIDCData,
  cdiRate: number,
  managerActive: boolean
): FIDCHealthReport {
  // 1. Subordination ratio (higher = healthier)
  const subordinationRatio = fidc.aum > 0
    ? (fidc.subordinatedValue / fidc.aum) * 100
    : 0;
  const subordinationScore = Math.min(100, subordinationRatio * 5); // 20%+ = 100

  // 2. Default rate (lower = healthier)
  const defaultRate = fidc.totalPortfolio > 0
    ? (fidc.defaultedCredits / fidc.totalPortfolio) * 100
    : 0;
  const defaultScore = Math.max(0, 100 - defaultRate * 10); // 10%+ = 0

  // 3. PDD coverage (higher = healthier)
  const pddCoverage = fidc.defaultedCredits > 0
    ? (fidc.pddProvisions / fidc.defaultedCredits) * 100
    : 100; // No defaults = full coverage
  const pddScore = Math.min(100, pddCoverage);

  // 4. Concentration (simplified - would need debtor data for Herfindahl)
  const concentrationScore = 70; // Default mid-range, enhance with Table VII data

  // 5. Yield spread vs CDI (anomaly detection)
  const annualizedReturn = fidc.monthlyReturn * 12;
  const spread = annualizedReturn - cdiRate;
  // Extreme returns (>50% spread) are suspicious
  const yieldScore = spread > 50 ? 20 : spread > 30 ? 50 : 80;

  // 6. Manager status
  const managerScore = managerActive ? 100 : 0;

  // Weighted composite
  const healthScore = Math.round(
    subordinationScore * WEIGHTS.subordination +
    defaultScore * WEIGHTS.defaultRate +
    pddScore * WEIGHTS.pddCoverage +
    concentrationScore * WEIGHTS.concentration +
    yieldScore * WEIGHTS.yieldSpread +
    managerScore * WEIGHTS.managerStatus
  );

  // Risk level thresholds
  const riskLevel: RiskLevel =
    healthScore <= 25 ? "CRITICAL" :
    healthScore <= 50 ? "HIGH" :
    healthScore <= 75 ? "MEDIUM" : "LOW";

  return {
    cnpj: fidc.cnpj,
    healthScore,
    riskLevel,
    subordinationRatio: Math.round(subordinationRatio * 100), // basis points
    defaultRate: Math.round(defaultRate * 100), // basis points
  };
}

contracts/FIDCHealthOracle.sol

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@chainlink/contracts/src/v0.8/shared/interfaces/AggregatorV3Interface.sol";

/**
 * @title FIDCHealthOracle
 * @notice Receives FIDC health reports from CRE workflow
 * @dev Implements IReceiver pattern for CRE integration
 */
contract FIDCHealthOracle {
    // ============ Types ============
    enum RiskLevel { LOW, MEDIUM, HIGH, CRITICAL }

    struct HealthReport {
        uint8 healthScore;           // 0-100
        RiskLevel riskLevel;
        uint256 subordinationRatio;  // basis points
        uint256 defaultRate;         // basis points
        uint256 aumUSD;              // AUM in USD (via Data Feed)
        uint256 lastUpdated;
    }

    // ============ State ============
    mapping(bytes32 => HealthReport) public reports;  // keccak256(cnpj) => report
    address public creForwarder;                    // CRE forwarder address
    AggregatorV3Interface public brlUsdFeed;         // Chainlink Data Feed

    // ============ Events ============
    event HealthReportUpdated(
        string indexed cnpj,
        uint8 healthScore,
        RiskLevel riskLevel,
        uint256 timestamp
    );

    event RiskAlert(
        string indexed cnpj,
        uint8 healthScore,
        RiskLevel riskLevel
    );

    // ============ Modifiers ============
    modifier onlyCREForwarder() {
        require(msg.sender == creForwarder, "Only CRE forwarder");
        _;
    }

    // ============ Constructor ============
    constructor(address _creForwarder, address _brlUsdFeed) {
        creForwarder = _creForwarder;
        brlUsdFeed = AggregatorV3Interface(_brlUsdFeed);
    }

    // ============ CRE Receiver ============
    /**
     * @notice Receives health report from CRE workflow
     * @param cnpj The FIDC CNPJ identifier
     * @param healthScore Composite health score (0-100)
     * @param riskLevel Risk classification
     * @param subordinationRatio Subordination ratio in basis points
     * @param defaultRate Default rate in basis points
     * @param aumBRL AUM in Brazilian Reais
     */
    function updateHealth(
        string calldata cnpj,
        uint8 healthScore,
        uint8 riskLevel,
        uint256 subordinationRatio,
        uint256 defaultRate,
        uint256 aumBRL
    ) external onlyCREForwarder {
        bytes32 key = keccak256(abi.encodePacked(cnpj));

        // Convert AUM to USD using Data Feed
        uint256 aumUSD = convertToUSD(aumBRL);

        reports[key] = HealthReport({
            healthScore: healthScore,
            riskLevel: RiskLevel(riskLevel),
            subordinationRatio: subordinationRatio,
            defaultRate: defaultRate,
            aumUSD: aumUSD,
            lastUpdated: block.timestamp
        });

        emit HealthReportUpdated(cnpj, healthScore, RiskLevel(riskLevel), block.timestamp);

        // Emit alert for HIGH or CRITICAL
        if (riskLevel >= uint8(RiskLevel.HIGH)) {
            emit RiskAlert(cnpj, healthScore, RiskLevel(riskLevel));
        }
    }

    // ============ Data Feed Integration ============
    function convertToUSD(uint256 amountBRL) internal view returns (uint256) {
        (, int256 price, , , ) = brlUsdFeed.latestRoundData();
        require(price > 0, "Invalid price");
        // BRL/USD typically has 8 decimals
        return (amountBRL * uint256(price)) / 1e8;
    }

    // ============ View Functions ============
    function getReport(string calldata cnpj) external view returns (HealthReport memory) {
        return reports[keccak256(abi.encodePacked(cnpj))];
    }

    /**
     * @notice PoR-style getter - treats subordination as reserve ratio
     * @dev Allows project to reference PoR design pattern in narrative
     */
    function getSubordinationAsReserveRatio(string calldata cnpj)
        external view returns (uint256)
    {
        return reports[keccak256(abi.encodePacked(cnpj))].subordinationRatio;
    }
}

contracts/RiskReceiver.sol (Arbitrum Sepolia)

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@chainlink/contracts-ccip/src/v0.8/ccip/applications/CCIPReceiver.sol";

/**
 * @title RiskReceiver
 * @notice Receives cross-chain risk alerts via CCIP
 */
contract RiskReceiver is CCIPReceiver {
    event CrossChainRiskAlert(
        uint64 indexed sourceChainSelector,
        string cnpj,
        uint8 healthScore,
        string alertType
    );

    struct AlertRecord {
        uint64 sourceChain;
        string cnpj;
        uint8 healthScore;
        uint256 timestamp;
    }

    AlertRecord[] public alertHistory;

    constructor(address router) CCIPReceiver(router) {}

    function _ccipReceive(
        Client.Any2EVMMessage memory message
    ) internal override {
        (string memory cnpj, uint8 healthScore, string memory alertType) =
            abi.decode(message.data, (string, uint8, string));

        alertHistory.push(AlertRecord({
            sourceChain: message.sourceChainSelector,
            cnpj: cnpj,
            healthScore: healthScore,
            timestamp: block.timestamp
        }));

        emit CrossChainRiskAlert(
            message.sourceChainSelector,
            cnpj,
            healthScore,
            alertType
        );
    }

    function getAlertCount() external view returns (uint256) {
        return alertHistory.length;
    }
}

import { Runner, HTTPClient, EVMClient } from "@chainlink/cre-sdk";
import { z } from "zod";
import { fetchCVMData } from "./fetchers/cvm";
import { fetchCDIRate } from "./fetchers/bcb";
import { verifyCNPJ } from "./fetchers/brasilapi";
import { computeHealthScore } from "./metrics/healthScore";
import type { FIDCHealthReport, Config } from "./types";

// Zod schema for config validation
const ConfigSchema = z.object({
  fidcCnpjs: z.array(z.string()),
  oracleAddress: z.string(),
  receiverAddress: z.string(),
  sepoliaRpcUrl: z.string(),
  arbitrumRpcUrl: z.string(),
  ccipRouterAddress: z.string(),
  dataFeedAddress: z.string(),  // BRL/USD
});

export default Runner.newRunner<z.infer<typeof ConfigSchema>>()
  .withConfigSchema(ConfigSchema)
  .withHandler(async (runtime, config, trigger) => {
    // 1. Fetch external data
    const cvmData = await runtime.sendRequest(
      async (http: HTTPClient) => fetchCVMData(http, config.fidcCnpjs)
    );

    const cdiRate = await runtime.sendRequest(
      async (http: HTTPClient) => fetchCDIRate(http)
    );

    const managerStatuses = await runtime.sendRequest(
      async (http: HTTPClient) =>
        Promise.all(cvmData.map(f => verifyCNPJ(http, f.managerCnpj)))
    );

    // 2. Compute health metrics for each FIDC
    const reports: FIDCHealthReport[] = cvmData.map((fidc, i) =>
      computeHealthScore(fidc, cdiRate, managerStatuses[i])
    );

    // 3. Generate and submit reports on-chain
    for (const report of reports) {
      // Write to FIDCHealthOracle.sol
      await runtime.report({
        targetChain: "ethereum-sepolia",
        targetAddress: config.oracleAddress,
        data: {
          cnpj: report.cnpj,
          healthScore: report.healthScore,
          riskLevel: report.riskLevel,
          subordinationRatio: report.subordinationRatio,
          defaultRate: report.defaultRate,
          timestamp: Date.now(),
        },
      });

      // 4. If CRITICAL, send CCIP alert to Arbitrum
      if (report.riskLevel === "CRITICAL") {
        await runtime.sendCCIP({
          destinationChain: "arbitrum-sepolia",
          receiver: config.receiverAddress,
          data: {
            sourceCnpj: report.cnpj,
            healthScore: report.healthScore,
            alertType: "CRITICAL_HEALTH",
          },
        });
      }
    }

    return { processed: reports.length, criticalAlerts: reports.filter(r => r.riskLevel === "CRITICAL").length };
  });

src/fetchers/cvm.ts - CVM Data Fetcher

import type { HTTPClient } from "@chainlink/cre-sdk";
import type { CVMFIDCData } from "../types";

const CVM_BASE_URL = "https://dados.cvm.gov.br/dados/FIDC/DOC/INF_MENSAL/DADOS";

export async function fetchCVMData(
  http: HTTPClient,
  cnpjs: string[]
): Promise<CVMFIDCData[]> {
  // Get latest monthly file (format: inf_mensal_fidc_YYYYMM.csv)
  const now = new Date();
  const yearMonth = `${now.getFullYear()}${String(now.getMonth()).padStart(2, '0')}`;
  const url = `${CVM_BASE_URL}/inf_mensal_fidc_${yearMonth}.csv`;

  const response = await http.get(url, {
    headers: { "Accept": "text/csv" },
  });

  if (!response.ok) {
    // Fallback to previous month
    const prevMonth = `${now.getFullYear()}${String(now.getMonth() - 1).padStart(2, '0')}`;
    const fallbackUrl = `${CVM_BASE_URL}/inf_mensal_fidc_${prevMonth}.csv`;
    const fallbackResp = await http.get(fallbackUrl);
    if (!fallbackResp.ok) throw new Error(`CVM fetch failed: ${fallbackResp.status}`);
    return parseCSV(fallbackResp.body, cnpjs);
  }

  return parseCSV(response.body, cnpjs);
}

function parseCSV(csv: string, filterCnpjs: string[]): CVMFIDCData[] {
  const lines = csv.split('\n');
  const headers = lines[0].split(';');

  // Key column indices (based on CVM schema)
  const cnpjIdx = headers.indexOf('CNPJ_FUNDO');
  const plIdx = headers.indexOf('VL_PATRIM_LIQ');           // AUM
  const subordinadaIdx = headers.indexOf('VL_COTA_SUBORDINADA');
  const inadimpIdx = headers.indexOf('VL_CARTEIRA_INADIMPLENTE');
  const carteiraIdx = headers.indexOf('VL_CARTEIRA_TOTAL');
  const pddIdx = headers.indexOf('VL_PROVISAO');
  const gestorIdx = headers.indexOf('CNPJ_GESTOR');
  const rentabIdx = headers.indexOf('RENTAB_MES');

  return lines.slice(1)
    .map(line => line.split(';'))
    .filter(cols => filterCnpjs.includes(cols[cnpjIdx]))
    .map(cols => ({
      cnpj: cols[cnpjIdx],
      aum: parseFloat(cols[plIdx]) || 0,
      subordinatedValue: parseFloat(cols[subordinadaIdx]) || 0,
      defaultedCredits: parseFloat(cols[inadimpIdx]) || 0,
      totalPortfolio: parseFloat(cols[carteiraIdx]) || 0,
      pddProvisions: parseFloat(cols[pddIdx]) || 0,
      managerCnpj: cols[gestorIdx],
      monthlyReturn: parseFloat(cols[rentabIdx]) || 0,
    }));
}

src/metrics/healthScore.ts - Composite Score Calculator

import type { CVMFIDCData, FIDCHealthReport, RiskLevel } from "../types";

const WEIGHTS = {
  subordination: 0.25,
  defaultRate: 0.25,
  pddCoverage: 0.20,
  concentration: 0.15,
  yieldSpread: 0.10,
  managerStatus: 0.05,
};

export function computeHealthScore(
  fidc: CVMFIDCData,
  cdiRate: number,
  managerActive: boolean
): FIDCHealthReport {
  // 1. Subordination ratio (higher = healthier)
  const subordinationRatio = fidc.aum > 0
    ? (fidc.subordinatedValue / fidc.aum) * 100
    : 0;
  const subordinationScore = Math.min(100, subordinationRatio * 5); // 20%+ = 100

  // 2. Default rate (lower = healthier)
  const defaultRate = fidc.totalPortfolio > 0
    ? (fidc.defaultedCredits / fidc.totalPortfolio) * 100
    : 0;
  const defaultScore = Math.max(0, 100 - defaultRate * 10); // 10%+ = 0

  // 3. PDD coverage (higher = healthier)
  const pddCoverage = fidc.defaultedCredits > 0
    ? (fidc.pddProvisions / fidc.defaultedCredits) * 100
    : 100; // No defaults = full coverage
  const pddScore = Math.min(100, pddCoverage);

  // 4. Concentration (simplified - would need debtor data for Herfindahl)
  const concentrationScore = 70; // Default mid-range, enhance with Table VII data

  // 5. Yield spread vs CDI (anomaly detection)
  const annualizedReturn = fidc.monthlyReturn * 12;
  const spread = annualizedReturn - cdiRate;
  // Extreme returns (>50% spread) are suspicious
  const yieldScore = spread > 50 ? 20 : spread > 30 ? 50 : 80;

  // 6. Manager status
  const managerScore = managerActive ? 100 : 0;

  // Weighted composite
  const healthScore = Math.round(
    subordinationScore * WEIGHTS.subordination +
    defaultScore * WEIGHTS.defaultRate +
    pddScore * WEIGHTS.pddCoverage +
    concentrationScore * WEIGHTS.concentration +
    yieldScore * WEIGHTS.yieldSpread +
    managerScore * WEIGHTS.managerStatus
  );

  // Risk level thresholds
  const riskLevel: RiskLevel =
    healthScore <= 25 ? "CRITICAL" :
    healthScore <= 50 ? "HIGH" :
    healthScore <= 75 ? "MEDIUM" : "LOW";

  return {
    cnpj: fidc.cnpj,
    healthScore,
    riskLevel,
    subordinationRatio: Math.round(subordinationRatio * 100), // basis points
    defaultRate: Math.round(defaultRate * 100), // basis points
  };
}

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@chainlink/contracts/src/v0.8/shared/interfaces/AggregatorV3Interface.sol";

/**
 * @title FIDCHealthOracle
 * @notice Receives FIDC health reports from CRE workflow
 * @dev Implements IReceiver pattern for CRE integration
 */
contract FIDCHealthOracle {
    // ============ Types ============
    enum RiskLevel { LOW, MEDIUM, HIGH, CRITICAL }

    struct HealthReport {
        uint8 healthScore;           // 0-100
        RiskLevel riskLevel;
        uint256 subordinationRatio;  // basis points
        uint256 defaultRate;         // basis points
        uint256 aumUSD;              // AUM in USD (via Data Feed)
        uint256 lastUpdated;
    }

    // ============ State ============
    mapping(bytes32 => HealthReport) public reports;  // keccak256(cnpj) => report
    address public creForwarder;                    // CRE forwarder address
    AggregatorV3Interface public brlUsdFeed;         // Chainlink Data Feed

    // ============ Events ============
    event HealthReportUpdated(
        string indexed cnpj,
        uint8 healthScore,
        RiskLevel riskLevel,
        uint256 timestamp
    );

    event RiskAlert(
        string indexed cnpj,
        uint8 healthScore,
        RiskLevel riskLevel
    );

    // ============ Modifiers ============
    modifier onlyCREForwarder() {
        require(msg.sender == creForwarder, "Only CRE forwarder");
        _;
    }

    // ============ Constructor ============
    constructor(address _creForwarder, address _brlUsdFeed) {
        creForwarder = _creForwarder;
        brlUsdFeed = AggregatorV3Interface(_brlUsdFeed);
    }

    // ============ CRE Receiver ============
    /**
     * @notice Receives health report from CRE workflow
     * @param cnpj The FIDC CNPJ identifier
     * @param healthScore Composite health score (0-100)
     * @param riskLevel Risk classification
     * @param subordinationRatio Subordination ratio in basis points
     * @param defaultRate Default rate in basis points
     * @param aumBRL AUM in Brazilian Reais
     */
    function updateHealth(
        string calldata cnpj,
        uint8 healthScore,
        uint8 riskLevel,
        uint256 subordinationRatio,
        uint256 defaultRate,
        uint256 aumBRL
    ) external onlyCREForwarder {
        bytes32 key = keccak256(abi.encodePacked(cnpj));

        // Convert AUM to USD using Data Feed
        uint256 aumUSD = convertToUSD(aumBRL);

        reports[key] = HealthReport({
            healthScore: healthScore,
            riskLevel: RiskLevel(riskLevel),
            subordinationRatio: subordinationRatio,
            defaultRate: defaultRate,
            aumUSD: aumUSD,
            lastUpdated: block.timestamp
        });

        emit HealthReportUpdated(cnpj, healthScore, RiskLevel(riskLevel), block.timestamp);

        // Emit alert for HIGH or CRITICAL
        if (riskLevel >= uint8(RiskLevel.HIGH)) {
            emit RiskAlert(cnpj, healthScore, RiskLevel(riskLevel));
        }
    }

    // ============ Data Feed Integration ============
    function convertToUSD(uint256 amountBRL) internal view returns (uint256) {
        (, int256 price, , , ) = brlUsdFeed.latestRoundData();
        require(price > 0, "Invalid price");
        // BRL/USD typically has 8 decimals
        return (amountBRL * uint256(price)) / 1e8;
    }

    // ============ View Functions ============
    function getReport(string calldata cnpj) external view returns (HealthReport memory) {
        return reports[keccak256(abi.encodePacked(cnpj))];
    }

    /**
     * @notice PoR-style getter - treats subordination as reserve ratio
     * @dev Allows project to reference PoR design pattern in narrative
     */
    function getSubordinationAsReserveRatio(string calldata cnpj)
        external view returns (uint256)
    {
        return reports[keccak256(abi.encodePacked(cnpj))].subordinationRatio;
    }
}

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@chainlink/contracts-ccip/src/v0.8/ccip/applications/CCIPReceiver.sol";

/**
 * @title RiskReceiver
 * @notice Receives cross-chain risk alerts via CCIP
 */
contract RiskReceiver is CCIPReceiver {
    event CrossChainRiskAlert(
        uint64 indexed sourceChainSelector,
        string cnpj,
        uint8 healthScore,
        string alertType
    );

    struct AlertRecord {
        uint64 sourceChain;
        string cnpj;
        uint8 healthScore;
        uint256 timestamp;
    }

    AlertRecord[] public alertHistory;

    constructor(address router) CCIPReceiver(router) {}

    function _ccipReceive(
        Client.Any2EVMMessage memory message
    ) internal override {
        (string memory cnpj, uint8 healthScore, string memory alertType) =
            abi.decode(message.data, (string, uint8, string));

        alertHistory.push(AlertRecord({
            sourceChain: message.sourceChainSelector,
            cnpj: cnpj,
            healthScore: healthScore,
            timestamp: block.timestamp
        }));

        emit CrossChainRiskAlert(
            message.sourceChainSelector,
            cnpj,
            healthScore,
            alertType
        );
    }

    function getAlertCount() external view returns (uint256) {
        return alertHistory.length;
    }
}
