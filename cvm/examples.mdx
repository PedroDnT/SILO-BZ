## Basic Examples

### Get Active Funds

Retrieve all currently active FIDC funds:

<CodeGroup>

```python Python
import requests

API_KEY = "your_api_key_here"
BASE_URL = "https://api.financialdata.com.br"

response = requests.get(
    f"{BASE_URL}/api/v1/cvm/fidc/cadastral",
    headers={"X-API-Key": API_KEY},
    params={"status": "ATIVO", "page_size": 100}
)

funds = response.json()['data']
print(f"Active FIDC funds: {len(funds)}")

for fund in funds[:5]:
    print(f"- {fund['DENOM_SOCIAL']} ({fund['CNPJ_FUNDO']})")
```

```javascript JavaScript

});
```

</CodeGroup>

## Advanced Examples

### Fetch All Pages

Retrieve all records across multiple pages:

```python
def fetch_all_pages(url, params=None):
    """Fetch all pages of data"""
    all_data = []
    page = 1
    
    while True:
        current_params = {**(params or {}), "page": page, "page_size": 1000}
        response = requests.get(
            url,
            headers={"X-API-Key": API_KEY},
            params=current_params
        )
        
        result = response.json()
        all_data.extend(result['data'])
        
        # Check if we've reached the last page
        pagination = result.get('pagination', {})
        if page >= pagination.get('total_pages', 1):
            break
            
        page += 1
        
    return all_data

# Get all active FIDC funds
all_fidcs = fetch_all_pages(
    f"{BASE_URL}/api/v1/cvm/fidc/cadastral",
    params={"status": "ATIVO"}
)

print(f"Total active FIDCs: {len(all_fidcs)}")
```

### Historical Data Analysis

Analyze historical trends over multiple months:

```python
from datetime import datetime, timedelta
import pandas as pd

def get_historical_data(entity, doc_type, months=12):
    """Get historical monthly data"""
    all_data = []
    current_date = datetime.now()
    
    for i in range(months):
        date = current_date - timedelta(days=30 * i)
        
        try:
            response = requests.get(
                f"{BASE_URL}/api/v1/cvm/{entity}/{doc_type}",
                headers={"X-API-Key": API_KEY},
                params={
                    "year": date.year,
                    "month": date.month,
                    "page_size": 1000
                }
            )
            
            data = response.json()['data']
            # Add period information
            for record in data:
                record['period'] = f"{date.year}-{date.month:02d}"
            
            all_data.extend(data)
            print(f"Fetched {len(data)} records for {date.year}-{date.month:02d}")
            
        except Exception as e:
            print(f"Error fetching data for {date.year}-{date.month:02d}: {e}")
            continue
    
    return all_data

# Get 12 months of FIDC data
historical_data = get_historical_data("fidc", "mensal", 12)

# Convert to DataFrame for analysis
df = pd.DataFrame(historical_data)
print(f"\nTotal records: {len(df)}")
print(f"Periods covered: {df['period'].nunique()}")
```

### Fund Performance Tracking

Track specific funds over time:

```python
def track_fund_performance(cnpj, months=6):
    """Track a specific fund's performance over time"""
    performance = []
    current_date = datetime.now()
    
    for i in range(months):
        date = current_date - timedelta(days=30 * i)
        
        response = requests.get(
            f"{BASE_URL}/api/v1/cvm/fidc/mensal",
            headers={"X-API-Key": API_KEY},
            params={
                "year": date.year,
                "month": date.month,
                "page_size": 1000
            }
        )
        
        data = response.json()['data']
        
        # Find the specific fund
        fund_data = next(
            (item for item in data if item.get('CNPJ_FUNDO') == cnpj),
            None
        )
        
        if fund_data:
            performance.append({
                'period': f"{date.year}-{date.month:02d}",
                'data': fund_data
            })
    
    return performance

# Track a specific fund
cnpj = "12.345.678/0001-90"
performance = track_fund_performance(cnpj, 6)

print(f"Performance data for {cnpj}:")
for period in performance:
    print(f"  {period['period']}: {period['data'].get('VL_PATRIMONIO', 'N/A')}")
```

### Export to CSV

Export data to CSV format:

```python
import csv

def export_to_csv(data, filename):
    """Export data to CSV file"""
    if not data:
        print("No data to export")
        return
    
    # Get all unique keys from all records
    fieldnames = set()
    for record in data:
        fieldnames.update(record.keys())
    
    fieldnames = sorted(fieldnames)
    
    with open(filename, 'w', newline='', encoding='utf-8') as csvfile:
        writer = csv.DictWriter(csvfile, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(data)
    
    print(f"Exported {len(data)} records to {filename}")

# Export active FIDCs to CSV
active_fidcs = fetch_all_pages(
    f"{BASE_URL}/api/v1/cvm/fidc/cadastral",
    params={"status": "ATIVO"}
)

export_to_csv(active_fidcs, "active_fidcs.csv")
```

### Data Aggregation

Aggregate data by different dimensions:

```python
from collections import defaultdict

def aggregate_by_status(entity="fidc"):
    """Aggregate funds by status"""
    response = requests.get(
        f"{BASE_URL}/api/v1/cvm/{entity}/cadastral",
        headers={"X-API-Key": API_KEY},
        params={"page_size": 1000}
    )
    
    funds = response.json()['data']
    
    # Aggregate by status
    by_status = defaultdict(int)
    for fund in funds:
        status = fund.get('SIT', 'UNKNOWN')
        by_status[status] += 1
    
    return dict(by_status)

# Get aggregation
status_counts = aggregate_by_status("fidc")
print("Funds by status:")
for status, count in status_counts.items():
    print(f"  {status}: {count}")
```

### Error Handling

Robust error handling for production use:

```python
import time
from typing import Optional, Dict, Any

class CVMAPIClient:
    def __init__(self, api_key: str, base_url: str = "https://api.financialdata.com.br"):
        self.api_key = api_key
        self.base_url = base_url
        self.session = requests.Session()
        self.session.headers.update({"X-API-Key": api_key})
    
    def get_data(
        self,
        entity: str,
        doc_type: str,
        params: Optional[Dict[str, Any]] = None,
        max_retries: int = 3
    ) -> Dict[str, Any]:
        """Get data with retry logic and error handling"""
        url = f"{self.base_url}/api/v1/cvm/{entity}/{doc_type}"
        
        for attempt in range(max_retries):
            try:
                response = self.session.get(url, params=params, timeout=30)
                
                if response.status_code == 200:
                    return response.json()
                
                elif response.status_code == 429:
                    # Rate limited
                    retry_after = int(response.headers.get('Retry-After', 60))
                    print(f"Rate limited. Waiting {retry_after} seconds...")
                    time.sleep(retry_after)
                    continue
                
                elif response.status_code == 503:
                    # Service unavailable
                    wait_time = 2 ** attempt
                    print(f"Service unavailable. Retrying in {wait_time} seconds...")
                    time.sleep(wait_time)
                    continue
                
                else:
                    response.raise_for_status()
                    
            except requests.exceptions.RequestException as e:
                if attempt < max_retries - 1:
                    wait_time = 2 ** attempt
                    print(f"Request failed: {e}. Retrying in {wait_time} seconds...")
                    time.sleep(wait_time)
                else:
                    raise
        
        raise Exception(f"Failed after {max_retries} attempts")

# Usage
client = CVMAPIClient(API_KEY)

try:
    data = client.get_data("fidc", "cadastral", params={"status": "ATIVO"})
    print(f"Retrieved {len(data['data'])} records")
except Exception as e:
    print(f"Error: {e}")
```

## Real-World Use Cases

### Portfolio Monitoring Dashboard

```python
def create_portfolio_summary(cnpj_list):
    """Create a summary for a portfolio of funds"""
    summary = []
    
    for cnpj in cnpj_list:
        # Get fund registration data
        response = requests.get(
            f"{BASE_URL}/api/v1/cvm/fidc/cadastral",
            headers={"X-API-Key": API_KEY},
            params={"page_size": 1000}
        )
        
        funds = response.json()['data']
        fund = next((f for f in funds if f['CNPJ_FUNDO'] == cnpj), None)
        
        if fund:
            summary.append({
                'cnpj': cnpj,
                'name': fund['DENOM_SOCIAL'],
                'status': fund['SIT'],
                'registration_date': fund['DT_REG']
            })
    
    return summary

# Monitor a portfolio
portfolio = [
    "12.345.678/0001-90",
    "98.765.432/0001-10"
]

portfolio_summary = create_portfolio_summary(portfolio)
for fund in portfolio_summary:
    print(f"{fund['name']}: {fund['status']}")
```
